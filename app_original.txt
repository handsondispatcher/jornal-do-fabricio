def update():
    with lock:
        if state['running']:return
        state['running']=True;state['message']='Coletando canais. Aguarde; a página recarregará automaticamente.'
    try:
        r=collect();state['message']=f'Coleta finalizada: {len(r["sources_ok"])} canais responderam, {len(r["sources_failed"])} falharam; {r["inserted"]} novos registros.'
    except Exception as e:state['message']='Falha na coleta: '+str(e)
    finally:state['running']=False
class Handler(BaseHTTPRequestHandler):
    def send_payload(self,body,mime='text/html; charset=utf-8',filename=None):
        self.send_response(200);self.send_header('Content-Type',mime);self.send_header('Cache-Control','no-store');self.send_header('X-Content-Type-Options','nosniff');self.send_header('Content-Security-Policy',"default-src 'none'; img-src 'self' https:; style-src 'unsafe-inline'; script-src 'unsafe-inline'; form-action 'self'; base-uri 'none'")
        if filename:self.send_header('Content-Disposition',f'attachment; filename="{filename}"')
        self.send_header('Content-Length',str(len(body)));self.end_headers();self.wfile.write(body)
    def do_GET(self):
        path=urlparse(self.path).path
        if path.startswith('/thumb/'):
            name=path.split('/')[-1]
            if not re.fullmatch(r'[0-9a-f]{64}\.(?:jpg|png|webp)',name):self.send_error(404);return
            file=BASE/'thumbnails'/name
            if not file.is_file():self.send_error(404);return
            self.send_payload(file.read_bytes(),mimetypes.guess_type(name)[0] or 'application/octet-stream');return
        if path=='/':self.send_payload(page(self.path).encode('utf-8'));return
        if path=='/exportar.csv':
            data=report();buf=io.StringIO();w=csv.writer(buf);w.writerow(['Título','Fonte','Publicado UTC','URL','Editorias','Tipo de descoberta','Resumo','Status'])
            for a in data['articles']:w.writerow([a['title'],a['source'],a['published'],a['url'],', '.join(a['sectors']),a['kind'],a['summary'],'NÃO CERTIFICADO'])
            self.send_payload(('\ufeff'+buf.getvalue()).encode('utf-8'),'text/csv; charset=utf-8','jornal_do_fabricio_export.csv');return
        self.send_error(404)
    def do_POST(self):
        if urlparse(self.path).path!='/atualizar':self.send_error(404);return
        with lock: running=state['running']
        if not running:threading.Thread(target=update,daemon=True).start()
        self.send_response(303);self.send_header('Location','/');self.send_header('Content-Length','0');self.end_headers()
if __name__=='__main__':
    server=ThreadingHTTPServer((HOST,PORT),Handler)
    print(f'Jornal do Fabricio v8: http://{HOST}:{PORT}/',flush=True)
    threading.Thread(target=update,daemon=True).start()
    threading.Timer(.7,lambda:webbrowser.open(f'http://{HOST}:{PORT}/')).start()
    try:server.serve_forever()
    except KeyboardInterrupt:pass
    finally:server.server_close()
